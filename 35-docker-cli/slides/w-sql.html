-- 1. 데이터베이스 생성

DROP DATABASE IF EXISTS kant_shop;

 

CREATE DATABASE kant_shop;

 

-- 2. 새로 만든 데이터베이스에 접속 (특정 DB를 사용하겠다.)

\c kant_shop

 

-- 3. 상품 테이블 생성

CREATE TABLE product

(

id SERIAL PRIMARY KEY,

name VARCHAR(100) NOT NULL,

price INT NOT NULL,

stock INT NOT NULL

);

 

-- 4. 상품 초기 데이터 입력

INSERT INTO product (name, price, stock) VALUES

('무선 이어폰', 35000, 100),

('게이밍 마우스', 28000, 85),

('27인치 모니터', 259000, 15),

('USB-C 허브', 19900, 200),

('노트북 거치대', 24000, 7);

있으면 지우고, 없으면 넘어가요

\c kant_shop

psql 명령이에요. 이 데이터베이스로 옮겨요

SERIAL · NOT NULL

id는 1, 2, 3 자동 번호, 나머지는 비울 수 없어요

INSERT … VALUES

상품 다섯 줄을 한 번에 넣어요

수업 파일 kant_shop.sql은 데이터베이스와 표를 만들고 상품 다섯 개를 넣어요. 하나씩 읽어 봐요. 하이픈 두 개로 시작하는 줄은 주석, 사람이 읽는 설명이라 실행하지 않아요. DROP DATABASE IF EXISTS는 kant_shop이 있으면 지우고, 없으면 그냥 넘어가요. 그래서 이 파일은 몇 번을 실행해도 같은 결과가 나와요. 백슬래시 c는 34에서 본 psql 명령이에요. 방금 만든 kant_shop으로 옮겨 가요. 표 product의 id는 SERIAL이라 1, 2, 3처럼 번호가 자동으로 붙어요. NOT NULL은 그 칸을 비워 둘 수 없다는 뜻이에요. 마지막 INSERT는 상품 다섯 줄을 한 번에 넣어요. 이 파일을 컨테이너 안의 psql에게 건네는 방법을 다음 장부터 봐요.

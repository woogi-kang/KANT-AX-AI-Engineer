menu = {"coffee": 4500, "latte": 5000, "cake": 6500, "tea": 4000}

stock = {"coffee": 3, "latte": 0, "cake": 2, "tea": 5}

print(list(filter(lambda m: stock[m] > 0, menu)))

print(list(filter(lambda m: stock[m] > 0 and menu[m] < 5000, menu)))

def add(self, product, qty):

# 품절 상품은 담을 수 없다.

if product.is_sold_out():

print("품절된 상품입니다.")

return

 

# 이미 장바구니에 담겨있는 수량

in_cart_qty = self.items.get(product, 0)

# 더 담을 수 있는 수량

extra_qty = product.stock - in_cart_qty

if qty > extra_qty: # 재고 수량 범위를 벋어난 경우

print(f"재고가 부족합니다. (담을 수 있는 수량:{extra_qty}개)")

return

 

self.items[product] = self.items.get(product, 0) + qty

print(f"{product.name} {qty}개를 담았습니다.")

add()의 return을 한 칸 단계 밖으로 꺼낸 코드예요. 이제 return은 if 상자 밖, add 상자 바로 안에 있어요. 텀블러는 품절이 아니라서 if 안의 print는 건너뛰어요. 그런데 return은 if와 상관없이 언제나 실행돼서 add()가 바로 끝나요. 그 아래 담는 코드까지 가지 못하니 장바구니가 비어 있다고 나와요. return을 if 안으로 4칸 넣으면, 품절일 때만 멈춰요. 19에서 return을 만나면 함수가 바로 끝난다고 배웠던 그 규칙이에요. 출력은 tools/demo.sh로 실제로 실행한 결과예요.

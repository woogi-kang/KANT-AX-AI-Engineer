class Stall:

def __init__(self, flavor, price, stock):

self.flavor = flavor

self.price = price

self.stock = stock

self.sold = 0

 

def sell(self, qty):

if qty > self.stock:

return f"{self.flavor} 재고가 모자라요"

self.stock = self.stock - qty

self.sold = self.sold + qty

return f"{self.flavor} {qty}개 판매"

 

def __str__(self):

return f"{self.flavor}: 판매 {self.sold}개, 매출 {self.sold * self.price:,}원"

Python

__init__: 재료 셋, 판매는 0

sell: if로 막고 상태 바꾸기

__str__: 매출 글씨

출력이 없는 셀이에요. 틀을 만들기만 했어요.

노점의 하루를 두 장에 걸쳐 만들어요. 먼저 틀이에요. 한 덩어리씩 짚어 봐요. __init__은 맛, 가격, 재고를 받고 판매 개수는 0에서 시작해요. sell은 4파트처럼 재고를 확인하고, 팔 수 있으면 상태를 바꾼 뒤 글자를 return해요. __str__은 판매 개수와 가격을 곱한 매출을 천 단위 콤마로 보여 줘요. 이 셀은 틀만 만들어서 출력이 없어요. 이 파트는 새 커널이라 셀 번호가 [1]부터예요. 출력은 실제로 실행한 결과예요.

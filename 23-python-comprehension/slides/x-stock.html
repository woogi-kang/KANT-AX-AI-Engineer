menu = {"coffee": 4500, "latte": 5000, "cake": 6500, "tea": 4000}

stock = {"coffee": 3, "latte": 0, "cake": 2, "tea": 0}

soldout = [m for m in stock if stock[m] == 0]

print("품절:", ", ".join(soldout))

board = {m: menu[m] * 11 // 10 for m in menu if stock[m] > 0}

print(board)

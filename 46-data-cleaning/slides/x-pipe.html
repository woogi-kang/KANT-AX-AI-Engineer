import pandas as pd

import seaborn as sns

 

df = sns.load_dataset("titanic")

 

df.loc[:, "age"] = df["age"].fillna(df["age"].mean())

town = df["embark_town"].mode()[0]

df.loc[:, "embark_town"] = df["embark_town"].fillna(town)

df.loc[:, "embarked"] = df["embark_town"].str[0]

df["deck"] = df["deck"].cat.add_categories("Unknown").fillna("Unknown")

df = df[df["fare"] < 512].copy()

 

print("남은 빈칸:", df.isnull().sum().sum())

print("크기:", df.shape)

Python

남은 빈칸: 0

크기: (888, 15)

← 숫자 빈칸: 평균

← 글자 빈칸: 최빈값

← 다른 열에서: 첫 글자

← category: 칸 늘리고

← 튀는 값: 512만 지우기

← 빈칸 0, 891행에서 888행

수업에서 셀마다 나눠 한 정리를 한 셀에 모았어요. 노트북을 새로 열어도 이 셀 하나로 같은 표를 다시 만들 수 있어요. 순서는 숫자 빈칸, 글자 빈칸, 다른 열에서 채우기, category, 튀는 값 지우기예요. 마지막 두 줄로 확인해요. isnull().sum()은 열마다 빈칸 수이고, 한 번 더 sum을 붙이면 표 전체의 빈칸 수예요. 0이고 크기는 888행 15열이에요. 다음 강의에서는 이 표를 모델에 넣어요. 이 셀만 새 노트북에서 실행해서 셀 번호가 [1]부터예요. 출력은 실제로 실행한 결과예요.

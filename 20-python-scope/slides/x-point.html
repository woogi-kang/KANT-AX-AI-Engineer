적립 함수: get과 기본값으로 포인트 쌓기

[4]

def add_point(points, name, amount, rate=5):

points[name] = points.get(name, 0) + amount * rate // 100

 

points = {"kim": 300}

add_point(points, "kim", 9000)

add_point(points, "lee", 6500)

add_point(points, "lee", 10000, rate=10)

print(points)

1. BEGIN, COMMIT, ROLLBACK

ROLLBACK은 BEGIN 뒤의 문장을 한꺼번에 되돌려요

kant_shop=# BEGIN;

BEGIN

kant_shop=*# UPDATE courses SET capacity = 30 WHERE course_id = 5;

UPDATE 1

kant_shop=*# DELETE FROM enrollments WHERE enroll_month = '2026-09';

DELETE 64

kant_shop=*# ROLLBACK;

ROLLBACK

kant_shop=# SELECT capacity FROM courses WHERE course_id = 5;

capacity

----------

      25

(1 row)

kant_shop=# SELECT COUNT(*) FROM enrollments WHERE enroll_month = '2026-09';

count

-------

   64

(1 row)
